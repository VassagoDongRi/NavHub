0. 通用约定与心法

项目 约定
下标 图论、树状数组、线段树、差分、背包：从 1 开始；快排/归并/二分：从 0 开始
数组大小 按上限 +5 开，防止 r+1、i+1 越界
无穷大 用 0x3f3f3f3f（约 1e9），两个相加不溢出 int
中间结果 乘法、计数、区间和一律用 long long
递归 深度可能到 n（如快排已排序）时考虑随机化或改迭代

写板子的三个固定动作

  1. 先写递归出口 / 初始化；
  2. 再写主循环；
  3. 最后检查"下标边界、开闭区间、是否要下传标记"。

1. 排序

1.1 快速排序

思路：选基准 pivot，双指针从两端向中间扫，把小于基准的甩到左边、大于的甩到右边，相遇后对两段分别递归。

复杂度：平均 O(n log n)，最坏 O(n²)（数组已排序且取端点做基准），额外空间 O(log n)（递归栈）。不稳定。

void quickSortUP(int a[], int l, int r) {          // 升序
	if (l >= r) return;                          // ① 递归出口
	int i = l, j = r, pivot = a[(l + r) / 2];    // ② 指针 + 基准
	while (i <= j) {                             // ③ 外层循环
		while (a[i] < pivot) i++;                // ④ 左指针右移（反转 4、5 步的比较号 → 降序）
		while (a[j] > pivot) j--;                // ⑤ 右指针左移
		if (i <= j) {                            // ⑥ 找到一对逆序，交换
			int t = a[i]; a[i] = a[j]; a[j] = t;
			i++; j--;                            // ⑦ 交换后两指针向内收
		}
	}
	quickSortUP(a, l, j);                          // ⑧ 递归左段
	quickSortUP(a, i, r);                          // ⑨ 递归右段
}

易错点

  • 递归出口是 l >= r，不是 l > r（一个元素也要停）。
  • 递归区间是 [l, j] 和 [i, r]，不是 [l, i]/[j, r]，否则会死循环。
  • 数据基本有序时把基准改成 a[l + rand() % (r - l + 1)] 或三数取中，防退化。

1.2 归并排序 + 求逆序对

思路：分治到单元素，合并两个有序段时统计"右边先出"造成的逆序。

复杂度：稳定 O(n log n)，额外空间 O(n)。

static long long msortInv(int a[], int tmp[], int l, int r) {
	if (l >= r) return 0;
	int mid = (l + r) >> 1;
	long long cnt = msortInv(a, tmp, l, mid) + msortInv(a, tmp, mid + 1, r);
	int i = l, j = mid + 1, k = l;
	while (i <= mid && j <= r) {
		if (a[i] <= a[j]) tmp[k++] = a[i++];       // 取等号才是稳定排序
		else { tmp[k++] = a[j++]; cnt += mid - i + 1; }  // 左段剩下全和它构成逆序
	}
	while (i <= mid) tmp[k++] = a[i++];
	while (j <= r)   tmp[k++] = a[j++];
	for (i = l; i <= r; i++) a[i] = tmp[i];
	return cnt;
}

long long mergeSortInv(int a[], int n) {           // 返回值 = 逆序对个数
	if (n <= 1) return 0;
	int *tmp = (int *)malloc(sizeof(int) * n);
	long long cnt = msortInv(a, tmp, 0, n - 1);
	free(tmp);
	return cnt;
}

易错点

  • 逆序对个数最大约 n(n-1)/2，n = 1e5 时约 5e9，必须用 long long。
  • cnt += mid - i + 1 的含义：右段 a[j] 先出，左段从 i 到 mid 都严格大于它。

2. 二分查找

2.1 整数二分（左闭右开）

记住一个模板就够：找"第一个满足条件的位置"，l 是答案，r 是永远不满足的哨兵。

int lowerBound(int a[], int n, int x) {            // 第一个 >= x 的下标（找不到返回 n）
	int l = 0, r = n;                            // [l, r)
	while (l < r) {
		int mid = l + (r - l) / 2;               // 防溢出
		if (a[mid] >= x) r = mid;                // 答案在左半边（含 mid）
		else l = mid + 1;                        // 答案在右半边
	}
	return l;
}

int upperBound(int a[], int n, int x) {            // 第一个 > x（找不到返回 n）
	int l = 0, r = n;
	while (l < r) {
		int mid = l + (r - l) / 2;
		if (a[mid] > x) r = mid;
		else l = mid + 1;
	}
	return l;
}

易错点

  • mid = (l + r) / 2 在 l、r 接近 2e9 时溢出，写 l + (r - l) / 2。
  • 用左闭右开就全程别混用闭区间，r 初始是 n 不是 n - 1。
  • 想找"最后一个 ≤ x"：用 upperBound(...) - 1。

2.2 实数二分 / 答案二分

// 实数二分：固定迭代 100 次，比 while (r - l > eps) 更稳，不会因精度死循环
double l = 0, r = 1e9;
for (int t = 0; t < 100; t++) {
	double mid = (l + r) / 2;
	if (check(mid)) r = mid;      // mid 可行，答案在左边
	else l = mid;                 // mid 不可行，答案在右边
}

答案二分（最大值最小 / 最小值最大）的套路：把"求最优值"改写成"判定 mid 是否可行"，check 必须是单调的（mid 可行则更大的 mid 也可行，或反之）。

易错点

  • check 不单调就不能二分，先证单调性。
  • 二分答案的区间要包含所有可能答案（含端点）。

3. 并查集

思路：每个集合用一棵树表示，根是代表元；查询时把路径上的点直接挂到根上（路径压缩），合并时把秩小的树挂到秩大的树下（按秩合并）。

复杂度：单次操作近似 O(1)（反阿克曼函数），n 次操作 O(n α(n))。

int fa[MAXN], rk[MAXN];

void dsuInit(int n) {
	for (int i = 1; i <= n; i++) { fa[i] = i; rk[i] = 0; }
}

int dsuFind(int x) {                               // 查根 + 路径压缩
	return fa[x] == x ? x : fa[x] = dsuFind(fa[x]);
}

void dsuUnion(int x, int y) {                      // 秩小的挂到秩大的下面
	x = dsuFind(x); y = dsuFind(y);
	if (x == y) return;                          // 已同集合
	if (rk[x] < rk[y]) { int t = x; x = y; y = t; }
	fa[y] = x;
	if (rk[x] == rk[y]) rk[x]++;
}

易错点

  • 合并前必须 dsuFind，不能直接 fa[x] = y。
  • 初始化时 fa[i] = i，从 0 开始还是 1 开始要和题目统一。

常见变体

  • 带权并查集：额外维护 dis[x] 表示 x 到根的距离，find 时累加。
  • 种类并查集：开 2n 或 3n 个点表示"敌人/朋友"（如食物链）。
  • 应用：Kruskal 判环、连通块计数、离线查询。

4. 树状数组

思路：bit[i] 存的是区间 (i - lowbit(i), i] 的和，改一个点向上跳，查前缀向下跳。

复杂度：单点修改 / 前缀和均 O(log n)，空间 O(n)。

int bit[MAXN], bitN;

void bitInit(int n) {
	bitN = n;
	memset(bit, 0, sizeof(int) * (n + 1));
}

void bitAdd(int i, int v) {                        // 第 i 位加 v
	for (; i <= bitN; i += i & -i) bit[i] += v;   // lowbit = i & -i，向上跳
}

int bitSum(int i) {                                // 1..i 的和
	int s = 0;
	for (; i > 0; i -= i & -i) s += bit[i];       // 向下跳
	return s;
}

int bitRange(int l, int r) {                       // l..r 的和
	return bitSum(r) - bitSum(l - 1);
}

易错点

  • 下标必须从 1 开始，bit[0] 用不到（i & -i 在 i=0 时为 0，会死循环）。
  • 上溯用 i += lowbit，下溯用 i -= lowbit，别写反。

常见变体

需求 做法
区间加、单点查 对差分数组建树状数组
区间加、区间查 维护两个树状数组（d[i] 和 i·d[i]）
求逆序对 离散化后从后往前，答案累加 bitSum(a[i]-1)
求第 k 小 树状数组上倍增（O(log n)）

5. 线段树

思路：每个节点管一段区间，区间修改时若被完全覆盖就打懒标记（不往下走），下次访问子节点前把标记下传。

复杂度：建树 O(n)，区间加 / 区间查 O(log n)，空间 4n。

long long seg[MAXN << 2], lazy[MAXN << 2];         // 开 4 倍

void segBuild(int p, int l, int r, int a[]) {
	lazy[p] = 0;
	if (l == r) { seg[p] = a[l]; return; }
	int mid = (l + r) >> 1;
	segBuild(p << 1, l, mid, a);
	segBuild(p << 1 | 1, mid + 1, r, a);
	seg[p] = seg[p << 1] + seg[p << 1 | 1];
}

void segPushDown(int p, int l, int r) {            // 标记下传
	if (lazy[p] == 0 || l == r) return;
	int mid = (l + r) >> 1;
	seg[p << 1]     += lazy[p] * (mid - l + 1);
	lazy[p << 1]    += lazy[p];
	seg[p << 1 | 1] += lazy[p] * (r - mid);
	lazy[p << 1 | 1] += lazy[p];
	lazy[p] = 0;
}

void segAdd(int p, int l, int r, int ql, int qr, long long v) {   // 区间加
	if (ql <= l && r <= qr) {                      // 完全覆盖
		seg[p] += v * (r - l + 1);
		lazy[p] += v;
		return;
	}
	segPushDown(p, l, r);                          // 部分覆盖：先下传
	int mid = (l + r) >> 1;
	if (ql <= mid) segAdd(p << 1, l, mid, ql, qr, v);
	if (qr > mid)  segAdd(p << 1 | 1, mid + 1, r, ql, qr, v);
	seg[p] = seg[p << 1] + seg[p << 1 | 1];
}

long long segQuery(int p, int l, int r, int ql, int qr) {         // 区间求和
	if (ql <= l && r <= qr) return seg[p];
	segPushDown(p, l, r);                          // 查询也要下传！
	int mid = (l + r) >> 1;
	long long res = 0;
	if (ql <= mid) res += segQuery(p << 1, l, mid, ql, qr);
	if (qr > mid)  res += segQuery(p << 1 | 1, mid + 1, r, ql, qr);
	return res;
}

易错点

  • 数组开 4n，开 2n 会越界。
  • pushDown 必须在递归子节点之前调用，segAdd 和 segQuery 都要。
  • 完全覆盖才打标记；部分覆盖必须递归后再 pushUp。
  • 多组数据时 build 要重新清零 lazy。

常见变体：区间乘（先乘后加，两个懒标记）、区间最值、单点修改（不需要 lazy）、动态开点 / 可持久化。

6. 图论：链式前向星 + Dijkstra

思路：head[u] 指向 u 的第一条边，每条边存 to、w、nxt（下一条边的编号）。Dijkstra 每次从堆里取当前距离最小的点，用它松弛邻边。

复杂度：O((n + m) log n)；要求边权非负。

typedef struct { int to, w, nxt; } Edge;
Edge e[MAXM];
int head[MAXN], ecnt;

void graphInit(void) {
	memset(head, 0, sizeof(head));
	ecnt = 0;
}

void addEdge(int u, int v, int w) {                // 无向图对 (u,v) 和 (v,u) 各调一次
	e[++ecnt].to  = v;
	e[ecnt].w    = w;
	e[ecnt].nxt  = head[u];
	head[u] = ecnt;
}

typedef struct { int d, u; } Node;                 // 距离、点
Node heap[MAXM + 5];
int hn;

void heapPush(Node x) {                            // 上浮
	int i = ++hn;
	heap[i] = x;
	while (i > 1 && heap[i].d < heap[i >> 1].d) {
		Node t = heap[i]; heap[i] = heap[i >> 1]; heap[i >> 1] = t;
		i >>= 1;
	}
}

Node heapPop(void) {                               // 取堆顶 + 下沉
	Node top = heap[1];
	heap[1] = heap[hn--];
	int i = 1;
	for (;;) {
		int s = i, l = i << 1, r = l | 1;
		if (l <= hn && heap[l].d < heap[s].d) s = l;
		if (r <= hn && heap[r].d < heap[s].d) s = r;
		if (s == i) break;
		Node t = heap[i]; heap[i] = heap[s]; heap[s] = t;
		i = s;
	}
	return top;
}

int dist[MAXN];

void dijkstra(int s, int n) {                      // 结果在 dist[1..n]
	for (int i = 1; i <= n; i++) dist[i] = INF;
	dist[s] = 0;
	hn = 0;
	heapPush((Node){0, s});
	while (hn > 0) {
		Node cur = heapPop();
		int u = cur.u, d = cur.d;
		if (d > dist[u]) continue;                // 过期元素，丢掉
		for (int i = head[u]; i != 0; i = e[i].nxt) {
			int v = e[i].to;
			if (dist[v] > d + e[i].w) {           // 松弛成功才入堆
				dist[v] = d + e[i].w;
				heapPush((Node){dist[v], v});
			}
		}
	}
}

易错点

  • C 没有 priority_queue，必须自己写堆；堆大小开"边数 + 5"。
  • 堆里可能有同一个点的多个旧距离，弹出时用 d > dist[u] 过滤。
  • 边权有负数时 Dijkstra 失效，改用 Bellman-Ford / SPFA（或 Johnson）。
  • 稠密图（m ≈ n²）用邻接矩阵 O(n²) 版本反而更快。

7. 字符串：KMP

思路：先对模式串求 nxt（nxt[i] = p[0..i] 的最长相等前后缀长度），匹配失配时模式串指针沿 nxt 回退，主串指针不回退。

复杂度：预处理 O(m)，匹配 O(n)，总 O(n + m)。

void kmpNext(const char p[], int nxt[]) {
	int m = (int)strlen(p);
	if (m == 0) return;
	nxt[0] = 0;
	for (int i = 1, j = 0; i < m; i++) {
		while (j > 0 && p[i] != p[j]) j = nxt[j - 1];
		if (p[i] == p[j]) j++;
		nxt[i] = j;
	}
}

int kmpCount(const char s[], const char p[], const int nxt[]) {   // 出现次数
	int n = (int)strlen(s), m = (int)strlen(p), cnt = 0;
	if (m == 0) return 0;
	for (int i = 0, j = 0; i < n; i++) {
		while (j > 0 && s[i] != p[j]) j = nxt[j - 1];
		if (s[i] == p[j]) j++;
		if (j == m) {
			cnt++;
			j = nxt[j - 1];                    // 允许重叠；不允许重叠写 j = 0
		}
	}
	return cnt;
}

易错点

  • nxt[0] = 0，且回退用 nxt[j - 1] 不是 nxt[j]（取决于定义，别混用两种写法）。
  • 匹配成功后要 j = nxt[j - 1] 继续找，否则漏掉重叠出现。
  • 求最小循环节：若 m % (m - nxt[m-1]) == 0，循环节长度是 m - nxt[m-1]。

8. 快速幂 / 快速乘

思路：把指数二进制拆分，底数不断平方，O(log b) 次乘法。

long long qpow(long long a, long long b, long long mod) {   // a^b % mod
	long long r = 1 % mod;                     // mod == 1 时结果 0
	while (b > 0) {
		if (b & 1) r = r * a % mod;
		a = a * a % mod;
		b >>= 1;
	}
	return r;
}

long long qmul(long long a, long long b, long long mod) {   // a*b % mod，防溢出
	long long r = 0;
	while (b > 0) {
		if (b & 1) r = (r + a) % mod;
		a = a * 2 % mod;
		b >>= 1;
	}
	return r;
}

易错点

  • 1 % mod 而不是 1，否则 mod = 1 时答案错。
  • mod 接近 1e18 时 a * a 直接溢出，改用 qmul 或 __int128。
  • 负指数要先求逆元（qpow(a, mod-2, mod) 当 mod 是素数）。

9. 01 背包

思路：dp[j] 表示容量 j 能装的最大价值，逐件物品考虑，容量倒序更新，保证每件只用一次。

复杂度：O(nW) 时间，O(W) 空间。

int knapDp[MAXN];                                  // 容量上限别超过 MAXN

int knap01(int w[], int v[], int n, int W) {       // w、v 从 1 开始
	memset(knapDp, 0, sizeof(int) * (W + 1));
	for (int i = 1; i <= n; i++) {
		for (int j = W; j >= w[i]; j--) {          // 倒序：保证一件最多用一次
			if (knapDp[j] < knapDp[j - w[i]] + v[i])
				knapDp[j] = knapDp[j - w[i]] + v[i];
		}
	}
	return knapDp[W];
}

易错点

  • 倒序写成正序就变成完全背包（每件无限用）。
  • "恰好装满"时初始化 dp[0] = 0，其余 -INF；"不超过容量"才全部初始化为 0。
  • 体积和价值都很大时考虑 meet-in-the-middle 或按价值 DP。

其余背包

类型 循环方向 备注
01 背包 容量倒序 每件 1 次
完全背包 容量正序 每件无限次
多重背包 二进制拆分 / 单调队列 第 i 件有 c 个
分组背包 先组、再容量倒序、再组内物品 每组最多选 1 个

10. 前缀和 / 差分

10.1 前缀和

long long pre[MAXN];                               // pre[i] = a[1] + ... + a[i]

void prefixSum(int a[], int n) {
	pre[0] = 0;
	for (int i = 1; i <= n; i++) pre[i] = pre[i - 1] + a[i];
}

long long rangeSum(int l, int r) {                 // O(1) 求 a[l..r]
	return pre[r] - pre[l - 1];
}

10.2 差分

用途：多次"区间整体加 v"，最后一次性还原。

int diff[MAXN + 2];                                // 开 n+2，防 r+1 越界

void diffBuild(int a[], int n) {                   // 由原数组构造差分
	memset(diff, 0, sizeof(int) * (n + 2));
	for (int i = 1; i <= n; i++) {
		diff[i]     += a[i];
		diff[i + 1] -= a[i];
	}
}

void diffAdd(int l, int r, int v) {                // a[l..r] 整体加 v，O(1)
	diff[l]     += v;
	diff[r + 1] -= v;
}

void diffToArray(int a[], int n) {                 // 前缀和还原
	int cur = 0;
	for (int i = 1; i <= n; i++) { cur += diff[i]; a[i] = cur; }
}

10.3 二维版（公式备查）

二维前缀和：
  pre[i][j] = a[i][j] + pre[i-1][j] + pre[i][j-1] - pre[i-1][j-1]

子矩阵 (x1,y1)~(x2,y2) 的和：
  pre[x2][y2] - pre[x1-1][y2] - pre[x2][y1-1] + pre[x1-1][y1-1]

二维差分（子矩阵整体加 v）：
  (x1,y1) += v, (x2+1,y1) -= v, (x1,y2+1) -= v, (x2+1,y2+1) += v
